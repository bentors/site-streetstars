// Regras utilizadas pro Firestore
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Produtos: leitura pública, escrita apenas admin
    match /products/{productId} {
      allow read: if true;
      allow write: if request.auth != null && 
                   get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.isAdmin == true;
    }
    
    // Pedidos: apenas o próprio usuário lê/escreve
    match /orders/{orderId} {
      allow read, write: if request.auth != null && 
                          request.auth.uid == resource.data.userId;
    }
    
    // Admins: apenas leitura por admins autenticados
    match /admins/{adminId} {
      allow read: if request.auth != null && 
                  request.auth.uid == adminId;
      allow write: if false; // Configure manualmente via console
    }
  }
}
